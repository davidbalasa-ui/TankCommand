import rawManifest from '../public/assets/manifest.json';
import { parseAssetManifest } from './assets/manifestSchema';
import { AssetRegistry } from './assets/registry';
import { createScreens } from './ui/screens';
import { ScreenRouter, type RendererHandle, type ScreenContext } from './ui/router';
import { createWorld, snapshot, step, type WorldState } from './sim/world';
import { validateCommand, type Command } from './sim/commands';
import { FixedStepLoop } from './sim/loop';
import { SIM_TICK_HZ } from './sim/constants';
import { Watchdog, installGlobalErrorCapture } from './platform/watchdog';
import { bindLifecycleToBrowserAndCapacitor, type Lifecycle } from './platform/lifecycle';
import type { RenderStats, RenderView } from './render/renderer';

export class App {
  public readonly router: ScreenRouter;
  private readonly registry: AssetRegistry;
  private world: WorldState;
  private pendingCommands: Command[] = [];
  private readonly fixed: FixedStepLoop;
  private readonly watchdog: Watchdog;
  private lifecycle: Lifecycle | null = null;
  private renderView: RenderView | null = null;
  private raf = 0;
  private checkTimer = 0;
  private lastFrame = 0;
  private paused = false;
  private uiPaused = false;
  private lifecyclePaused = false;
  private freezeSimUntil = 0;

  constructor(root: HTMLElement) {
    this.registry = new AssetRegistry(parseAssetManifest(rawManifest));
    this.world = createWorld(1);
    this.fixed = new FixedStepLoop({
      tickHz: SIM_TICK_HZ,
      maxStepsPerFrame: 5,
      step: () => {
        this.stepSimulation();
      },
    });
    this.watchdog = new Watchdog({
      now: () => performance.now(),
      freezeMs: 2000,
      recoverMs: 5000,
      onFreeze: () => {
        this.router.go('recovery');
      },
      onRecover: () => undefined,
    });

    this.router = new ScreenRouter(root, createScreens(), (router): ScreenContext => ({
      router,
      go: (id) => {
        router.go(id);
      },
      pauseSim: (paused) => {
        this.pauseSim(paused);
      },
      attachRenderer: (canvas) => this.attachRenderer(canvas),
    }));
  }

  start(): void {
    this.router.start('title');
    this.lifecycle = bindLifecycleToBrowserAndCapacitor();
    this.lifecycle.on('pause', () => {
      this.pauseLifecycle();
    });
    this.lifecycle.on('resume', () => {
      this.resumeLifecycle();
    });
    this.lifecycle.on('back', () => {
      this.router.back();
    });
    window.addEventListener('keydown', this.onKeyDown);
    installGlobalErrorCapture(window, (error) => {
      console.error(error.message);
    });
  }

  snapshot(): WorldState {
    return { ...snapshot(this.world) };
  }

  sendCommand(command: unknown): boolean {
    const validated = validateCommand(command);
    if (!validated) {
      return false;
    }
    this.pendingCommands.push(validated);
    return true;
  }

  renderStats(): RenderStats {
    if (!this.renderView) {
      return { loadedModels: [], meshCount: 0, drawCalls: 0, frame: 0 };
    }
    return this.renderView.stats();
  }

  freezeSim(ms: number): void {
    this.freezeSimUntil = performance.now() + Math.max(0, ms);
  }

  pauseSim(paused: boolean): void {
    this.uiPaused = paused;
    this.applyPauseState();
  }

  private attachRenderer(canvas: HTMLCanvasElement): Promise<RendererHandle> {
    return import('./render/renderer').then(
      (mod) => {
        const view = new mod.RenderView(canvas, this.registry);
        this.renderView = view;
        this.startLoop();
        return {
          error: null,
          dispose: (): void => {
            if (this.renderView === view) {
              this.stopLoop();
              view.dispose();
              this.renderView = null;
            }
          },
        };
      },
      () => ({
        error: '3D rendering is unavailable on this device.',
        dispose: (): void => undefined,
      }),
    );
  }

  private readonly onKeyDown = (event: KeyboardEvent): void => {
    if (event.key === 'Escape') {
      this.router.back();
    }
  };

  private pauseLifecycle(): void {
    this.lifecyclePaused = true;
    this.applyPauseState();
  }

  private resumeLifecycle(): void {
    this.lifecyclePaused = false;
    this.applyPauseState();
  }

  private applyPauseState(): void {
    const next = this.uiPaused || this.lifecyclePaused;
    if (next === this.paused) {
      return;
    }
    this.paused = next;
    if (next) {
      this.watchdog.setSuspended(true);
      this.stopLoop();
    } else {
      this.watchdog.setSuspended(false);
      this.startLoop();
    }
  }

  private startLoop(): void {
    if (this.raf !== 0 || !this.renderView || this.paused) {
      return;
    }
    this.lastFrame = performance.now();
    this.watchdog.beat('sim');
    this.watchdog.beat('render');
    this.checkTimer = window.setInterval(() => {
      this.watchdog.check();
    }, 250);
    this.raf = window.requestAnimationFrame(this.frame);
  }

  private stopLoop(): void {
    if (this.raf !== 0) {
      window.cancelAnimationFrame(this.raf);
      this.raf = 0;
    }
    if (this.checkTimer !== 0) {
      window.clearInterval(this.checkTimer);
      this.checkTimer = 0;
    }
  }

  private readonly frame = (now: number): void => {
    this.raf = 0;
    if (this.paused || !this.renderView) {
      return;
    }

    const elapsed = now - this.lastFrame;
    this.lastFrame = now;
    const frozen = now < this.freezeSimUntil;
    let alpha = 0;

    if (!frozen) {
      const result = this.fixed.advance(elapsed);
      alpha = result.alpha;
      if (result.steps > 0) {
        this.watchdog.beat('sim');
      }
    }

    this.watchdog.beat('render');
    this.watchdog.check();
    this.renderView.render(alpha);
    this.raf = window.requestAnimationFrame(this.frame);
  };

  private stepSimulation(): void {
    const nextTick = this.world.tick + 1;
    const due = this.pendingCommands.filter((command) => command.tick === nextTick);
    this.world = step(this.world, due);
    this.pendingCommands = this.pendingCommands.filter((command) => command.tick > nextTick);
  }
}

export function createApp(root: HTMLElement): App {
  return new App(root);
}

export interface KvStore {
  get(key: string): Promise<string | null>;
  set(key: string, value: string): Promise<void>;
  remove(key: string): Promise<void>;
  keys(): Promise<string[]>;
}

export class StorageFullError extends Error {
  constructor(message = 'Storage is full') {
    super(message);
    this.name = 'StorageFullError';
  }
}

function toError(error: unknown): Error {
  if (error instanceof Error) {
    return error;
  }
  return new Error(String(error));
}

export class MemoryKvStore implements KvStore {
  private readonly data = new Map<string, string>();

  get(key: string): Promise<string | null> {
    return Promise.resolve(this.data.get(key) ?? null);
  }

  set(key: string, value: string): Promise<void> {
    this.data.set(key, value);
    return Promise.resolve();
  }

  remove(key: string): Promise<void> {
    this.data.delete(key);
    return Promise.resolve();
  }

  keys(): Promise<string[]> {
    return Promise.resolve([...this.data.keys()].sort());
  }
}

function isQuotaError(error: unknown): boolean {
  if (error instanceof DOMException && (error.name === 'QuotaExceededError' || error.code === 22)) {
    return true;
  }
  if (error instanceof Error) {
    return (
      error.name === 'QuotaExceededError' || error.message.toLowerCase().includes('quota')
    );
  }
  return false;
}

export class WebKvStore implements KvStore {
  private readonly storage: Storage;

  constructor(storage?: Storage) {
    this.storage = storage ?? globalThis.localStorage;
  }

  get(key: string): Promise<string | null> {
    try {
      return Promise.resolve(this.storage.getItem(key));
    } catch (error) {
      return Promise.reject(toError(error));
    }
  }

  set(key: string, value: string): Promise<void> {
    try {
      this.storage.setItem(key, value);
      return Promise.resolve();
    } catch (error) {
      if (isQuotaError(error)) {
        return Promise.reject(new StorageFullError());
      }
      return Promise.reject(toError(error));
    }
  }

  remove(key: string): Promise<void> {
    try {
      this.storage.removeItem(key);
      return Promise.resolve();
    } catch (error) {
      return Promise.reject(toError(error));
    }
  }

  keys(): Promise<string[]> {
    try {
      const result: string[] = [];
      for (let i = 0; i < this.storage.length; i += 1) {
        const key = this.storage.key(i);
        if (key !== null) {
          result.push(key);
        }
      }
      return Promise.resolve(result);
    } catch (error) {
      return Promise.reject(toError(error));
    }
  }
}

const SETTINGS_PREFIX = 'setting.';

export class NativeKvStore implements KvStore {
  async get(key: string): Promise<string | null> {
    if (key.startsWith(SETTINGS_PREFIX)) {
      const result = await Preferences.get({ key });
      return result.value;
    }
    try {
      const result = await Filesystem.readFile({
        path: `saves/${key}.json`,
        directory: Directory.Data,
        encoding: Encoding.UTF8,
      });
      return typeof result.data === 'string' ? result.data : null;
    } catch {
      return null;
    }
  }

  async set(key: string, value: string): Promise<void> {
    if (key.startsWith(SETTINGS_PREFIX)) {
      await Preferences.set({ key, value });
      return;
    }
    await Filesystem.writeFile({
      path: `saves/${key}.json`,
      directory: Directory.Data,
      data: value,
      encoding: Encoding.UTF8,
    });
  }

  async remove(key: string): Promise<void> {
    if (key.startsWith(SETTINGS_PREFIX)) {
      await Preferences.remove({ key });
      return;
    }
    try {
      await Filesystem.deleteFile({
        path: `saves/${key}.json`,
        directory: Directory.Data,
      });
    } catch {
      return;
    }
  }

  async keys(): Promise<string[]> {
    if (!Capacitor.isNativePlatform()) {
      return [];
    }

    const result: string[] = [];
    try {
      const listing = await Filesystem.readdir({
        path: 'saves',
        directory: Directory.Data,
      });
      const entries = listing.files as unknown as Array<string | { name: string }>;
      for (const entry of entries) {
        const name = typeof entry === 'string' ? entry : entry.name;
        if (name.endsWith('.json')) {
          result.push(name.slice(0, -5));
        }
      }
    } catch {
      // The saves directory may not exist yet.
    }

    const preferences = await Preferences.keys();
    result.push(...preferences.keys);
    return result;
  }
}

export function createKvStore(): KvStore {
  return Capacitor.isNativePlatform() ? new NativeKvStore() : new WebKvStore();

export type WatchdogChannel = 'sim' | 'render';

export interface WatchdogReport {
  channel: WatchdogChannel;
  stalledMs: number;
  lastBeats: Record<WatchdogChannel, number>;
}

export interface WatchdogOptions {
  now: () => number;
  freezeMs?: number;
  recoverMs?: number;
  onFreeze: (report: WatchdogReport) => void;
  onRecover: (report: WatchdogReport) => void;
}

interface ChannelState {
  lastBeat: number;
  freezeFired: boolean;
  recoverFired: boolean;
}

export class Watchdog {
  private readonly opts: WatchdogOptions;
  private readonly freezeMs: number;
  private readonly recoverMs: number;
  private readonly states: Record<WatchdogChannel, ChannelState>;
  private suspended = false;

  constructor(opts: WatchdogOptions) {
    this.opts = opts;
    this.freezeMs = opts.freezeMs ?? 2000;
    this.recoverMs = opts.recoverMs ?? 5000;
    const now = opts.now();
    this.states = {
      sim: { lastBeat: now, freezeFired: false, recoverFired: false },
      render: { lastBeat: now, freezeFired: false, recoverFired: false },
    };
  }

  beat(channel: WatchdogChannel): void {
    const state = this.states[channel];
    state.lastBeat = this.opts.now();
    state.freezeFired = false;
    state.recoverFired = false;
  }

  setSuspended(suspended: boolean): void {
    this.suspended = suspended;
    if (!suspended) {
      const now = this.opts.now();
      for (const channel of ['sim', 'render'] as const) {
        const state = this.states[channel];
        state.lastBeat = now;
        state.freezeFired = false;
        state.recoverFired = false;
      }
    }
  }

  check(): void {
    if (this.suspended) {
      return;
    }

    const now = this.opts.now();
    const stalled = (['sim', 'render'] as const).map((channel) => {
      const state = this.states[channel];
      return {
        channel,
        state,
        stalledMs: now - state.lastBeat,
      };
    });

    const freeze = stalled.find(
      (item) => item.stalledMs >= this.freezeMs && !item.state.freezeFired,
    );
    if (freeze) {
      for (const item of stalled) {
        if (item.stalledMs >= this.freezeMs) {
          item.state.freezeFired = true;
        }
      }
      this.opts.onFreeze({
        channel: freeze.channel,
        stalledMs: freeze.stalledMs,
        lastBeats: {
          sim: this.states.sim.lastBeat,
          render: this.states.render.lastBeat,
        },
      });
    }

    const recover = stalled.find(
      (item) => item.stalledMs >= this.recoverMs && !item.state.recoverFired,
    );
    if (recover) {
      for (const item of stalled) {
        if (item.stalledMs >= this.recoverMs) {
          item.state.recoverFired = true;
        }
      }
      this.opts.onRecover({
        channel: recover.channel,
        stalledMs: recover.stalledMs,
        lastBeats: {
          sim: this.states.sim.lastBeat,
          render: this.states.render.lastBeat,
        },
      });
    }
  }
}

export interface CapturedError {
  message: string;
  stack?: string;
  kind: 'error' | 'unhandledrejection';
}

export interface ErrorTarget {
  addEventListener(type: string, listener: (event: Event) => void): void;
  removeEventListener?(type: string, listener: (event: Event) => void): void;
}

export function installGlobalErrorCapture(
  target: ErrorTarget,
  onError: (error: CapturedError) => void,
): () => void {
  const errorHandler = (event: Event): void => {
    const errorEvent = event as ErrorEvent;
    const rawError: unknown = errorEvent.error;
    const error = rawError instanceof Error ? rawError : undefined;
    const message =
      errorEvent.message ||
      error?.message ||
      (typeof rawError === 'string' ? rawError : 'Unknown error');
    onError({
      message,
      stack: error?.stack,
      kind: 'error',
    });
  };

  const rejectionHandler = (event: Event): void => {
    const rejection = event as PromiseRejectionEvent;
    const reason: unknown = rejection.reason;
    const error = reason instanceof Error ? reason : undefined;
    onError({
      message: error?.message ?? String(reason),
      stack: error?.stack,
      kind: 'unhandledrejection',
    });
  };

  target.addEventListener('error', errorHandler);
  target.addEventListener('unhandledrejection', rejectionHandler);

  return () => {
    target.removeEventListener?.('error', errorHandler);
    target.removeEventListener?.('unhandledrejection', rejectionHandler);
  };
}

const OVERLAY_IDS = new Set<ScreenId>(['pause', 'objectives', 'help', 'recovery', 'bugReport']);

const ALLOWED: Record<ScreenId, readonly ScreenId[]> = {
  title: ['campaignSelect', 'skirmishSetup', 'settings', 'credits', 'help'],
  campaignSelect: ['title', 'missionSelect'],
  missionSelect: ['campaignSelect', 'briefing'],
  briefing: ['missionSelect', 'loading'],
  skirmishSetup: ['title', 'loading'],
  loading: ['inGame', 'title'],
  inGame: ['pause', 'objectives', 'help', 'bugReport', 'recovery', 'results', 'title'],
  pause: ['settings', 'bugReport', 'title', 'inGame'],
  settings: ['title', 'pause', 'inGame'],
  help: ['title', 'inGame'],
  objectives: ['inGame'],
  results: ['title', 'missionSelect'],
  credits: ['title'],
  bugReport: ['title', 'inGame', 'pause'],
  recovery: ['inGame', 'title'],
};

const PARENT: Partial<Record<ScreenId, ScreenId>> = {
  campaignSelect: 'title',
  missionSelect: 'campaignSelect',
  briefing: 'missionSelect',
  skirmishSetup: 'title',
  loading: 'title',
  settings: 'title',
  help: 'title',
  objectives: 'inGame',
  results: 'title',
  credits: 'title',
  bugReport: 'inGame',
  recovery: 'inGame',
  pause: 'inGame',
};

export class ScreenRouter {
  private readonly root: HTMLElement;
  private readonly screens: Record<ScreenId, ScreenModule>;
  private readonly contextFactory: (router: ScreenRouter) => ScreenContext;
  private context: ScreenContext | null = null;
  private base: MountedScreen | null = null;
  private overlays: MountedScreen[] = [];

  constructor(
    root: HTMLElement,
    screens: Record<ScreenId, ScreenModule>,
    contextFactory: (router: ScreenRouter) => ScreenContext,
  ) {
    this.root = root;
    this.screens = screens;
    this.contextFactory = contextFactory;
  }

  start(id: ScreenId = 'title'): void {
    this.mountBase(id);
  }

  current(): ScreenId {
    return this.overlays.at(-1)?.id ?? this.base?.id ?? 'title';
  }

  go(id: ScreenId): void {
    const current = this.current();
    if (id === current) {
      return;
    }

    const overlay = this.isOverlayTarget(id, current);
    if (!this.canGo(current, id)) {
      this.reject(id);
      return;
    }

    if (overlay) {
      this.mountOverlay(id);
    } else {
      this.mountBase(id);
    }
  }

  back(): void {
    if (this.overlays.length > 0) {
      this.popOverlay();
      return;
    }
    if (!this.base) {
      return;
    }
    if (this.base.id === 'inGame') {
      this.go('pause');
      return;
    }
    const parent = PARENT[this.base.id];
    if (parent) {
      this.go(parent);
    }
  }

  private isOverlayTarget(id: ScreenId, current: ScreenId): boolean {
    if (id === 'settings') {
      return current === 'pause' && this.base?.id === 'inGame';
    }
    return OVERLAY_IDS.has(id) && this.base?.id === 'inGame';
  }

  private canGo(from: ScreenId, to: ScreenId): boolean {
    return ALLOWED[from].includes(to);
  }

  private reject(id: ScreenId): void {
    const message = `Screen transition not allowed: ${id}`;
    if (import.meta.env.DEV) {
      throw new Error(message);
    }
    console.warn(message);
  }

  private mountBase(id: ScreenId): void {
    for (const overlay of [...this.overlays].reverse()) {
      this.unmountScreen(overlay);
    }
    this.overlays = [];
    if (this.base) {
      this.unmountScreen(this.base);
    }
    const mounted = this.createScreen(id, false);
    mounted.el.style.visibility = 'visible';
    this.base = mounted;
  }

  private mountOverlay(id: ScreenId): void {
    const top = this.currentMounted();
    if (top) {
      top.el.style.visibility = 'hidden';
    }
    const mounted = this.createScreen(id, true);
    mounted.el.style.visibility = 'visible';
    this.overlays.push(mounted);
  }

  private popOverlay(): void {
    const mounted = this.overlays.pop();
    if (!mounted) {
      return;
    }
    this.unmountScreen(mounted);
    const top = this.currentMounted();
    if (top) {
      top.el.style.visibility = 'visible';
    }
  }

  private currentMounted(): MountedScreen | null {
    return this.overlays.at(-1) ?? this.base;
  }

  private createScreen(id: ScreenId, overlay: boolean): MountedScreen {
    const el = document.createElement('div');
    el.className = `screen screen-${id}${overlay ? ' overlay' : ''}`;
    el.dataset.screenId = id;
    this.root.append(el);
    const screen = this.screens[id];
    const unmount = screen.mount(el, this.ensureContext());
    return { id, el, unmount };
  }

  private unmountScreen(mounted: MountedScreen): void {
    mounted.unmount();
    mounted.el.remove();
  }

  private ensureContext(): ScreenContext {
    if (!this.context) {
      this.context = this.contextFactory(this);
    }
    return this.context;
  }
}

type Build = (root: HTMLElement, ctx: ScreenContext) => void | (() => void);

function makeScreen(id: ScreenId, title: string, build?: Build): ScreenModule {
  return {
    id,
    mount(root, ctx) {
      root.replaceChildren();
      const heading = document.createElement('h1');
      heading.textContent = title;
      root.append(heading);
      const cleanup = build?.(root, ctx);
      return () => {
        cleanup?.();
        root.replaceChildren();
      };
    },
  };
}

    for (let j = i + 1; j < boxes.length; j += 1) {
      const a = boxes[i];
      const b = boxes[j];

export function countDistinctPngColors(buffer: Buffer): number {
  const signature = buffer.subarray(0, 8).toString('hex');
  if (signature !== '89504e470d0a1a0a') {
    throw new Error('Not a PNG');
  }

  let offset = 8;
  let width = 0;
  let height = 0;
  let bitDepth = 0;
  let colorType = 0;
  const idat: Buffer[] = [];

  while (offset < buffer.length) {
    const length = buffer.readUInt32BE(offset);
    const type = buffer.toString('ascii', offset + 4, offset + 8);
    const data = buffer.subarray(offset + 8, offset + 8 + length);
    if (type === 'IHDR') {
      width = data.readUInt32BE(0);
      height = data.readUInt32BE(4);
      bitDepth = data[8] ?? 0;
      colorType = data[9] ?? 0;
    } else if (type === 'IDAT') {
      idat.push(data);
    } else if (type === 'IEND') {
      break;
    }
    offset += 12 + length;
  }

  if (bitDepth !== 8 || (colorType !== 6 && colorType !== 2)) {
    throw new Error(`Unsupported PNG format: bitDepth=${bitDepth} colorType=${colorType}`);
  }

  const bpp = colorType === 6 ? 4 : 3;
  const stride = width * bpp;
  const raw = inflateSync(Buffer.concat(idat));
  const out = Buffer.alloc(stride * height);
  let pos = 0;

  for (let y = 0; y < height; y += 1) {
    const filter = raw[pos];
    pos += 1;
    if (filter === undefined) {
      throw new Error('Truncated PNG');
    }
    const rowStart = y * stride;
    for (let x = 0; x < stride; x += 1) {
      const rawByte = raw[pos];
      pos += 1;
      if (rawByte === undefined) {
        throw new Error('Truncated PNG');
      }
      const left = x >= bpp ? out[rowStart + x - bpp] ?? 0 : 0;
      const up = y > 0 ? out[rowStart - stride + x] ?? 0 : 0;
      const upLeft = y > 0 && x >= bpp ? out[rowStart - stride + x - bpp] ?? 0 : 0;
      let value = 0;
      if (filter === 0) {
        value = rawByte;
      } else if (filter === 1) {
        value = rawByte + left;
      } else if (filter === 2) {
        value = rawByte + up;
      } else if (filter === 3) {
        value = rawByte + Math.floor((left + up) / 2);
      } else if (filter === 4) {
        const p = left + up - upLeft;
        const pa = Math.abs(p - left);
        const pb = Math.abs(p - up);
        const pc = Math.abs(p - upLeft);
        const predictor = pa <= pb && pa <= pc ? left : pb <= pc ? up : upLeft;
        value = rawByte + predictor;
      } else {
        throw new Error(`Unsupported PNG filter ${filter}`);
      }
      out[rowStart + x] = value & 0xff;
    }
  }

  const colors = new Set<string>();
  for (let y = 0; y < height; y += 1) {
    for (let x = 0; x < width; x += 1) {
      const index = y * stride + x * bpp;
      colors.add(`${out[index] ?? 0},${out[index + 1] ?? 0},${out[index + 2] ?? 0}`);
    }
  }
  return colors.size;
}

### 2. E2E determinism

- **`freezeSim` semantics decided.** `freezeSim(ms)` keeps its exact meaning: a timed freeze that suppresses sim steps **and** sim watchdog beats, used only by WD-04 to make the watchdog trip within a bounded window. It is not a general-purpose "pause".
- **New hook method `pauseSim(paused)`.** Added to `TestHook` (`src/testhook/testhook.ts`, `tests/e2e/helpers.ts`) and to `App` (`pauseSim(paused: boolean)`) which delegates to the same UI-pause path the pause screen now uses (`ScreenContext.pauseSim`). Internally `App` now tracks `uiPaused` and `lifecyclePaused` separately and derives the effective `paused` flag (`applyPauseState`), so the pause screen and the visibility lifecycle don't clobber each other. `pause()`’s `watchdog.setSuspended(true)` + `stopLoop()` and `resume()`’s `setSuspended(false)` + `startLoop()` are preserved.
- **`src/ui/screens.ts` pause screen** now actually pauses the sim: `ctx.pauseSim(true)` on mount, `ctx.pauseSim(false)` in its unmount cleanup (which the router runs when the pause overlay is popped or replaced). This makes the "same path the pause screen uses" true.
- **HOOK-02 rewritten** to use `pauseSim(true)` instead of `freezeSim`. Root cause of the flake: `freezeSim(1000)` only guaranteed "no sim steps for the next second"; each `page.evaluate` round-trip cost tens to hundreds of ms, so on some runs the 1 s budget expired mid-test, the sim resumed, ~2 extra ticks ran (counter 90 → 110), and the `afterInvalid === beforeInvalid` assertion tripped. With `pauseSim`, the sim is genuinely stopped (rAF cancelled, `paused` flag set), and the counter cannot move. Test amount was raised to 1000 so that even if the command were silently dropped the RNG-driven counter could not reach the threshold within the 5 s poll budget — the assertion is now strict, not permissive.
- **UI-05 root cause: stale watchdog beats.** After the async renderer chunk resolves and `startLoop()` runs, the watchdog constructor's `lastBeat` (set at page load) could already be >2 s old — the first frames run at ~60 Hz so no sim step (and thus no `beat('sim')`) happens until 50 ms elapsed. On the tablet viewport (higher resolution → slightly slower render startup) the very first `check()` crossed the 2 s threshold and `onFreeze` navigated to `recovery`, which hides the `inGame` base (`visibility: hidden`) and removed the Pause button from the accessibility tree before the click could land. Fix: `startLoop()` now calls `watchdog.beat('sim')` and `watchdog.beat('render')` before starting the interval, so every loop start (initial mount, resume from lifecycle pause, resume from UI pause) begins a fresh watchdog window.
