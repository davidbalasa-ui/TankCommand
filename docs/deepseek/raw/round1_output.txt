<<<FILE src/sim/rng.ts>>>
export class Rng {
  private state: number;

  constructor(seed: number) {
    this.state = seed >>> 0;
  }

  nextU32(): number {
    this.state = (this.state + 0x6d2b79f5) >>> 0;
    let t = this.state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return (t ^ (t >>> 14)) >>> 0;
  }

  next(): number {
    return this.nextU32() / 4294967296;
  }

  int(minInclusive: number, maxExclusive: number): number {
    return minInclusive + Math.floor(this.next() * (maxExclusive - minInclusive));
  }

  getState(): number {
    return this.state >>> 0;
  }

  setState(s: number): void {
    this.state = s >>> 0;
  }
}

import { Rng } from './rng';
import type { Command } from './commands';

export interface WorldState {
  tick: number;
  seed: number;
  rngState: number;
  counter: number;
}

export function createWorld(seed: number): WorldState {
  const rng = new Rng(seed);
  return {
    tick: 0,
    seed,
    rngState: rng.getState(),
    counter: 0,
  };
}

export function step(world: WorldState, commands: readonly Command[]): WorldState {
  const nextTick = world.tick + 1;
  let counter = world.counter;

  for (const command of commands) {
    if (command.kind === 'debug.addCounter' && command.tick === nextTick) {
      counter += command.amount;
    }
  }

  const rng = new Rng(world.seed);
  rng.setState(world.rngState);
  counter += rng.int(0, 3);

  return {
    tick: nextTick,
    seed: world.seed,
    rngState: rng.getState(),
    counter,
  };
}

export function snapshot(world: WorldState): Readonly<WorldState> {
  return Object.freeze({ ...world });
}

export function canonicalJson(value: unknown): string {
  if (Array.isArray(value)) {
    return `[${value.map((item) => canonicalJson(item)).join(',')}]`;
  }
  if (value !== null && typeof value === 'object') {
    const record = value as Record<string, unknown>;
    const keys = Object.keys(record).sort();
    return `{${keys
      .map((key) => `${JSON.stringify(key)}:${canonicalJson(record[key])}`)
      .join(',')}}`;
  }
  return JSON.stringify(value) ?? 'null';
}

export function fnv1aHex(input: string): string {
  let hash = 0x811c9dc5;
  for (let i = 0; i < input.length; i += 1) {
    hash ^= input.charCodeAt(i) & 0xff;
    hash = Math.imul(hash, 0x01000193) >>> 0;
  }
  return hash.toString(16).padStart(8, '0');
}

export function hashWorld(world: WorldState): string {
  return fnv1aHex(
    canonicalJson({
      tick: world.tick,
      seed: world.seed,
      rngState: world.rngState,
      counter: world.counter,
    }),
  );
}

export interface FixedStepResult {
  steps: number;
  alpha: number;
}

export interface FixedStepOptions {
  tickHz: number;
  maxStepsPerFrame: number;
  step: () => void;
}

export class FixedStepLoop {
  private accumulator = 0;
  private readonly tickMs: number;
  private readonly maxSteps: number;
  private readonly opts: FixedStepOptions;

  constructor(opts: FixedStepOptions) {
    if (!(opts.tickHz > 0)) {
      throw new Error('tickHz must be greater than zero');
    }
    this.opts = opts;
    this.tickMs = 1000 / opts.tickHz;
    this.maxSteps = Math.max(1, Math.floor(opts.maxStepsPerFrame));
  }

  advance(elapsedMs: number): FixedStepResult {
    const elapsed = Number.isFinite(elapsedMs) && elapsedMs > 0 ? elapsedMs : 0;
    this.accumulator += elapsed;

    const rawSteps = Math.floor(this.accumulator / this.tickMs);
    let steps = rawSteps;
    if (rawSteps > this.maxSteps) {
      steps = this.maxSteps;
      this.accumulator = 0;
    } else {
      this.accumulator -= steps * this.tickMs;
    }

    for (let i = 0; i < steps; i += 1) {
      this.opts.step();
    }

    const alpha = Math.min(Math.max(this.accumulator / this.tickMs, 0), 0.999999);
    return { steps, alpha };
  }
}

export const LICENSE_VALUES = ['CC0', 'CC-BY-4.0', 'CC-BY-3.0', 'ORIGINAL'] as const;
export type License = (typeof LICENSE_VALUES)[number];
export type AssetKind = 'model' | 'texture' | 'audio';

const assetEntrySchema = z.object({
  id: z.string().regex(/^[a-z0-9]+(\.[a-z0-9_]+)+$/),
  kind: z.enum(['model', 'texture', 'audio']),
  path: z.string(),
  bytes: z.number().int().positive(),
  faction: z.enum(['human', 'alien', 'neutral']).optional(),
  lod: z
    .object({
      level: z.union([z.literal(0), z.literal(1), z.literal(2)]),
      of: z.string(),
    })
    .optional(),
  tags: z.array(z.string()).optional(),
  credit: z.object({
    license: z.enum(LICENSE_VALUES),
    author: z.string().min(1),
    title: z.string().min(1),
    source: z.string().url().optional(),
    usedFor: z.string().min(1),
  }),
});

export type AssetEntry = z.infer<typeof assetEntrySchema>;

export interface AssetManifest {
  version: 1;
  assets: AssetEntry[];
}

export class AssetManifestError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'AssetManifestError';
  }
}

const topSchema = z.object({
  version: z.literal(1),
  assets: z.array(z.unknown()),
});

function extractId(raw: unknown): string {
  if (typeof raw === 'object' && raw !== null && 'id' in raw) {
    const value = (raw as { id?: unknown }).id;
    if (typeof value === 'string') {
      return value;
    }
  }
  return '<unknown>';
}

export function parseAssetManifest(input: unknown): AssetManifest {
  const top = topSchema.safeParse(input);
  if (!top.success) {
    throw new AssetManifestError(
      `Invalid manifest: ${top.error.issues.map((issue) => issue.message).join('; ')}`,
    );
  }

  const assets: AssetEntry[] = [];
  for (const raw of top.data.assets) {
    const id = extractId(raw);
    const parsed = assetEntrySchema.safeParse(raw);
    if (!parsed.success) {
      const issue = parsed.error.issues[0];
      throw new AssetManifestError(`asset "${id}": ${issue?.message ?? 'invalid'}`);
    }
    assets.push(parsed.data);
  }

  const ids = new Set<string>();
  const paths = new Set<string>();

  for (const entry of assets) {
    if (ids.has(entry.id)) {
      throw new AssetManifestError(`asset "${entry.id}": duplicate id`);
    }
    ids.add(entry.id);

    if (paths.has(entry.path)) {
      throw new AssetManifestError(`asset "${entry.id}": duplicate path "${entry.path}"`);
    }
    paths.add(entry.path);

    if (
      !entry.path.startsWith('assets/') ||
      entry.path.startsWith('/') ||
      entry.path.split('/').includes('..')
    ) {
      throw new AssetManifestError(`asset "${entry.id}": invalid path`);
    }

    if (entry.credit.license.startsWith('CC-BY') && !entry.credit.source) {
      throw new AssetManifestError(`asset "${entry.id}": CC-BY licence requires source`);
    }

    if (entry.credit.source && !entry.credit.source.startsWith('https://')) {
      throw new AssetManifestError(`asset "${entry.id}": source must be an https URL`);
    }
  }

  for (const entry of assets) {
    if (entry.lod) {
      if (entry.lod.of === entry.id) {
        throw new AssetManifestError(`asset "${entry.id}": lod.of cannot point to itself`);
      }
      if (!ids.has(entry.lod.of)) {
        throw new AssetManifestError(`asset "${entry.id}": lod.of "${entry.lod.of}" not found`);
      }
    }
  }

  return { version: 1, assets };
}

import type { AssetEntry, AssetManifest, AssetKind } from './manifestSchema';

export interface AssetFilter {
  kind?: AssetKind;
  faction?: 'human' | 'alien' | 'neutral';
  tag?: string;
}

export class UnknownAssetError extends Error {
  public readonly id: string;

  constructor(id: string) {
    super(`Unknown asset: ${id}`);
    this.name = 'UnknownAssetError';
    this.id = id;
  }
}

export class AssetRegistry {
  private readonly byId = new Map<string, AssetEntry>();

  constructor(manifest: AssetManifest) {
    for (const entry of manifest.assets) {
      this.byId.set(entry.id, entry);
    }
  }

  get(id: string): AssetEntry {
    const entry = this.byId.get(id);
    if (!entry) {
      throw new UnknownAssetError(id);
    }
    return entry;
  }

  list(filter: AssetFilter = {}): AssetEntry[] {
    return [...this.byId.values()].filter((entry) => {
      if (filter.kind && entry.kind !== filter.kind) {
        return false;
      }
      if (filter.faction && entry.faction !== filter.faction) {
        return false;
      }
      if (filter.tag && !(entry.tags ?? []).includes(filter.tag)) {
        return false;
      }
      return true;
    });
  }

  lodsOf(id: string): AssetEntry[] {
    this.get(id);
    return [...this.byId.values()]
      .filter((entry) => entry.lod?.of === id)
      .sort((a, b) => (a.lod?.level ?? 0) - (b.lod?.level ?? 0));
  }
}

import type { AssetEntry } from './manifestSchema';
import type { AssetRegistry } from './registry';

export type AssetDecoder<T> = (buffer: ArrayBuffer, entry: AssetEntry) => Promise<T>;

export interface AssetDecoders<M, T, A> {
  model: AssetDecoder<M>;
  texture: AssetDecoder<T>;
  audio: AssetDecoder<A>;
}

export type DecodedAsset<M, T, A> = M | T | A;

export interface AssetLoaderOptions<M, T, A> {
  registry: AssetRegistry;
  fetchBytes: (url: string) => Promise<ArrayBuffer>;
  decoders: AssetDecoders<M, T, A>;
  baseUrl: string;
  dispose?: (asset: DecodedAsset<M, T, A>) => void;
}

export interface PreloadProgress {
  loaded: number;
  total: number;
  id: string;
}

interface CacheEntry<M, T, A> {
  refCount: number;
  resolved: boolean;
  asset?: DecodedAsset<M, T, A>;
  promise: Promise<DecodedAsset<M, T, A>>;
}

export class AssetLoadError extends Error {
  public readonly id: string;
  public override readonly cause: unknown;

  constructor(id: string, cause: unknown) {
    super(`Failed to load asset "${id}"`);
    this.name = 'AssetLoadError';
    this.id = id;
    this.cause = cause;
  }
}

export class AssetLoader<M = unknown, T = unknown, A = unknown> {
  private readonly options: AssetLoaderOptions<M, T, A>;
  private readonly cache = new Map<string, CacheEntry<M, T, A>>();

  constructor(options: AssetLoaderOptions<M, T, A>) {
    this.options = options;
  }

  load(id: string): Promise<DecodedAsset<M, T, A>> {
    const entry = this.options.registry.get(id);
    const existing = this.cache.get(id);
    if (existing) {
      existing.refCount += 1;
      return existing.promise;
    }

    const cacheEntry: Partial<CacheEntry<M, T, A>> = {
      refCount: 1,
      resolved: false,
    };

    const promise = this.loadInternal(entry).then(
      (asset) => {
        cacheEntry.resolved = true;
        cacheEntry.asset = asset;
        return asset;
      },
      (error: unknown) => {
        this.cache.delete(id);
        if (error instanceof AssetLoadError) {
          throw error;
        }
        throw new AssetLoadError(id, error);
      },
    );

    cacheEntry.promise = promise;
    this.cache.set(id, cacheEntry as CacheEntry<M, T, A>);
    return promise;
  }

  release(id: string): void {
    const entry = this.cache.get(id);
    if (!entry) {
      return;
    }
    entry.refCount -= 1;
    if (entry.refCount > 0) {
      return;
    }
    this.cache.delete(id);
    if (entry.resolved) {
      this.options.dispose?.(entry.asset as DecodedAsset<M, T, A>);
    }
  }

  async preload(ids: readonly string[], onProgress?: (progress: PreloadProgress) => void): Promise<void> {
    const total = ids.length;
    let loaded = 0;
    const failures: Array<{ id: string; error: unknown }> = [];

    await Promise.all(
      ids.map(async (id) => {
        try {
          await this.load(id);
        } catch (error) {
          failures.push({ id, error });
        } finally {
          loaded += 1;
          onProgress?.({ loaded, total, id });
        }
      }),
    );

    if (failures.length > 0) {
      const message = `Failed to preload assets: ${failures.map((failure) => failure.id).join(', ')}`;
      throw new AggregateError(
        failures.map((failure) => failure.error),
        message,
      );
    }
  }

  isCached(id: string): boolean {
    return this.cache.has(id);
  }

  private async loadInternal(entry: AssetEntry): Promise<DecodedAsset<M, T, A>> {
    const url = this.urlFor(entry);
    const buffer = await this.options.fetchBytes(url);
    if (buffer.byteLength !== entry.bytes) {
      throw new Error(`size mismatch for ${entry.id}: expected ${entry.bytes}, got ${buffer.byteLength}`);
    }

    if (entry.kind === 'model') {
      return this.options.decoders.model(buffer, entry);
    }
    if (entry.kind === 'texture') {
      return this.options.decoders.texture(buffer, entry);
    }
    return this.options.decoders.audio(buffer, entry);
  }

  private urlFor(entry: AssetEntry): string {
    const base = this.options.baseUrl.endsWith('/')
      ? this.options.baseUrl
      : `${this.options.baseUrl}/`;
    return `${base}${entry.path}`;
  }
}

export function loadUnitTable(json: unknown, assetRegistry: AssetRegistry): ReadonlyMap<string, UnitRow> {
  if (!Array.isArray(json)) {
    throw new UnitDataError('Unit table must be an array');
  }

  const table = new Map<string, UnitRow>();

  for (const [index, raw] of json.entries()) {
    const parsed = unitRowSchema.safeParse(raw);
    if (!parsed.success) {
      const id =
        typeof raw === 'object' && raw !== null && 'id' in raw
          ? String((raw as { id?: unknown }).id)
          : `row ${index}`;
      throw new UnitDataError(`${id}: ${parsed.error.issues[0]?.message ?? 'invalid row'}`);
    }

    const row = parsed.data;
    if (table.has(row.id)) {
      throw new UnitDataError(`${row.id}: duplicate id`);
    }

    try {
      assetRegistry.get(row.modelAssetId);
    } catch {
      throw new UnitDataError(`${row.id}: missing model asset ${row.modelAssetId}`);
    }

    table.set(row.id, row);
  }

  return table;
}

  return [...table.values()]
    .filter((row) => row.source.kind === 'placeholder')
    .map((row) => row.id)
    .sort();

export interface KvStore {
  get(key: string): Promise<string | null>;
  set(key: string, value: string): Promise<void>;
  remove(key: string): Promise<void>;
  keys(): Promise<string[]>;
}

export class StorageFullError extends Error {
  constructor(message = 'Storage is full') {
    super(message);
    this.name = 'StorageFullError';
  }
}

export class MemoryKvStore implements KvStore {
  private readonly data = new Map<string, string>();

  async get(key: string): Promise<string | null> {
    return this.data.get(key) ?? null;
  }

  async set(key: string, value: string): Promise<void> {
    this.data.set(key, value);
  }

  async remove(key: string): Promise<void> {
    this.data.delete(key);
  }

  async keys(): Promise<string[]> {
    return [...this.data.keys()].sort();
  }
}

function isQuotaError(error: unknown): boolean {
  if (error instanceof DOMException && (error.name === 'QuotaExceededError' || error.code === 22)) {
    return true;
  }
  if (error instanceof Error) {
    return (
      error.name === 'QuotaExceededError' || error.message.toLowerCase().includes('quota')
    );
  }
  return false;
}

export class WebKvStore implements KvStore {
  private readonly storage: Storage;

  constructor(storage?: Storage) {
    this.storage = storage ?? globalThis.localStorage;
  }

  async get(key: string): Promise<string | null> {
    return this.storage.getItem(key);
  }

  async set(key: string, value: string): Promise<void> {
    try {
      this.storage.setItem(key, value);
    } catch (error) {
      if (isQuotaError(error)) {
        throw new StorageFullError();
      }
      throw error;
    }
  }

  async remove(key: string): Promise<void> {
    this.storage.removeItem(key);
  }

  async keys(): Promise<string[]> {
    const result: string[] = [];
    for (let i = 0; i < this.storage.length; i += 1) {
      const key = this.storage.key(i);
      if (key !== null) {
        result.push(key);
      }
    }
    return result;
  }
}

export class NativeKvStore implements KvStore {
  async get(key: string): Promise<string | null> {
    if (key.startsWith(SETTINGS_PREFIX)) {
      const result = await Preferences.get({ key });
      return result.value;
    }
    try {
      const result = await Filesystem.readFile({
        path: `saves/${key}.json`,
        directory: Directory.Data,
        encoding: Encoding.UTF8,
      });
      return typeof result.data === 'string' ? result.data : null;
    } catch {
      return null;
    }
  }

  async set(key: string, value: string): Promise<void> {
    if (key.startsWith(SETTINGS_PREFIX)) {
      await Preferences.set({ key, value });
      return;
    }
    await Filesystem.writeFile({
      path: `saves/${key}.json`,
      directory: Directory.Data,
      data: value,
      encoding: Encoding.UTF8,
    });
  }

  async remove(key: string): Promise<void> {
    if (key.startsWith(SETTINGS_PREFIX)) {
      await Preferences.remove({ key });
      return;
    }
    try {
      await Filesystem.deleteFile({
        path: `saves/${key}.json`,
        directory: Directory.Data,
      });
    } catch {
      return;
    }
  }

  async keys(): Promise<string[]> {
    if (!Capacitor.isNativePlatform()) {
      return [];
    }

    const result: string[] = [];
    try {
      const listing = await Filesystem.readdir({
        path: 'saves',
        directory: Directory.Data,
      });
      const entries = listing.files as unknown as Array<string | { name: string }>;
      for (const entry of entries) {
        const name = typeof entry === 'string' ? entry : entry.name;
        if (name.endsWith('.json')) {
          result.push(name.slice(0, -5));
        }
      }
    } catch {
      // The saves directory may not exist yet.
    }

    const preferences = await Preferences.keys();
    result.push(...preferences.keys);
    return result;
  }
}

import type { WorldState } from '../sim/world';
import { canonicalJson, fnv1aHex } from '../sim/world';
import type { KvStore } from './kv';
import { StorageFullError } from './kv';

export interface SaveMeta {
  gameVersion: string;
  savedAtIso: string;
}

export type DecodeFailureReason = 'corrupt' | 'checksum' | 'incompatible' | 'not-a-save';

export type DecodeSaveResult =
  | { ok: true; world: WorldState; meta: SaveMeta }
  | { ok: false; reason: DecodeFailureReason };

export interface LoadedSave {
  world: WorldState;
  meta: SaveMeta;
}

export const CURRENT_SAVE_SCHEMA = 1;
export const migrations: Record<number, (payload: unknown) => unknown> = {};

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null;
}

function isWorldState(value: unknown): value is WorldState {
  if (!isRecord(value)) {
    return false;
  }
  return (
    typeof value.tick === 'number' &&
    Number.isInteger(value.tick) &&
    typeof value.seed === 'number' &&
    Number.isFinite(value.seed) &&
    typeof value.rngState === 'number' &&
    Number.isFinite(value.rngState) &&
    typeof value.counter === 'number' &&
    Number.isFinite(value.counter)
  );
}

export function encodeSave(world: WorldState, meta: SaveMeta): string {
  const payload: WorldState = {
    tick: world.tick,
    seed: world.seed,
    rngState: world.rngState,
    counter: world.counter,
  };
  const checksum = fnv1aHex(canonicalJson(payload));
  return JSON.stringify({
    format: 'tcsave',
    schema: CURRENT_SAVE_SCHEMA,
    gameVersion: meta.gameVersion,
    savedAtIso: meta.savedAtIso,
    checksum,
    payload,
  });
}

export function decodeSave(text: string): DecodeSaveResult {
  try {
    const parsed: unknown = JSON.parse(text);
    if (!isRecord(parsed)) {
      return { ok: false, reason: 'not-a-save' };
    }
    if (parsed.format !== 'tcsave') {
      return { ok: false, reason: 'not-a-save' };
    }

    const schema = parsed.schema;
    if (typeof schema !== 'number' || !Number.isInteger(schema)) {
      return { ok: false, reason: 'corrupt' };
    }
    if (schema > CURRENT_SAVE_SCHEMA) {
      return { ok: false, reason: 'incompatible' };
    }

    let payload: unknown = parsed.payload;
    let version = schema;
    while (version < CURRENT_SAVE_SCHEMA) {
      const migrate = migrations[version];
      if (!migrate) {
        return { ok: false, reason: 'incompatible' };
      }
      payload = migrate(payload);
      version += 1;
    }

    const checksum = parsed.checksum;
    if (typeof checksum !== 'string') {
      return { ok: false, reason: 'corrupt' };
    }
    if (fnv1aHex(canonicalJson(payload)) !== checksum) {
      return { ok: false, reason: 'checksum' };
    }

    if (!isWorldState(payload)) {
      return { ok: false, reason: 'corrupt' };
    }

    const gameVersion = parsed.gameVersion;
    const savedAtIso = parsed.savedAtIso;
    if (typeof gameVersion !== 'string' || typeof savedAtIso !== 'string') {
      return { ok: false, reason: 'corrupt' };
    }

    return {
      ok: true,
      world: payload,
      meta: { gameVersion, savedAtIso },
    };
  } catch {
    return { ok: false, reason: 'corrupt' };
  }
}

export interface SaveManagerOptions {
  maxBytes?: number;
  gameVersion?: string;
  now?: () => Date;
}

const encoder = new TextEncoder();

function byteLength(text: string): number {
  return encoder.encode(text).byteLength;
}

export class SaveManager {
  private readonly kv: KvStore;
  private readonly maxBytes: number;
  private readonly gameVersion: string;
  private readonly now: () => Date;
  private readonly bytes = new Map<string, number>();
  private totalBytes = 0;
  private loaded = false;
  private autoCursor = 0;
  private lastAutoSlot: string | null = null;

  constructor(kv: KvStore, options: SaveManagerOptions = {}) {
    this.kv = kv;
    this.maxBytes = options.maxBytes ?? 5 * 1024 * 1024;
    this.gameVersion = options.gameVersion ?? '0.1.0';
    this.now = options.now ?? (() => new Date());
  }

  async save(slot: string, world: WorldState): Promise<void> {
    await this.ensureLoaded();
    const text = encodeSave(world, {
      gameVersion: this.gameVersion,
      savedAtIso: this.now().toISOString(),
    });
    const key = this.key(slot);
    const size = byteLength(text);
    const previous = this.bytes.get(key) ?? 0;

    if (this.totalBytes - previous + size > this.maxBytes) {
      throw new StorageFullError();
    }

    await this.kv.set(key, text);
    this.bytes.set(key, size);
    this.totalBytes = this.totalBytes - previous + size;
  }

  async load(slot: string): Promise<LoadedSave | null> {
    const text = await this.kv.get(this.key(slot));
    if (text === null) {
      return null;
    }
    const decoded = decodeSave(text);
    if (!decoded.ok) {
      throw new Error(`Cannot load save "${slot}": ${decoded.reason}`);
    }
    return { world: decoded.world, meta: decoded.meta };
  }

  async listSlots(): Promise<string[]> {
    await this.ensureLoaded();
    return [...this.bytes.keys()]
      .filter((key) => key.startsWith('save:'))
      .map((key) => key.slice('save:'.length))
      .sort();
  }

  async autosave(world: WorldState): Promise<void> {
    await this.ensureLoaded();
    const slot = `auto-${this.autoCursor % 3}`;
    await this.save(slot, world);
    this.lastAutoSlot = slot;
    this.autoCursor = (this.autoCursor + 1) % 3;
  }

  async latestAutosave(): Promise<LoadedSave | null> {
    await this.ensureLoaded();
    const slots = (await this.listSlots()).filter((slot) => slot.startsWith('auto-'));
    const first = slots[0];
    if (!first) {
      return null;
    }

    if (this.lastAutoSlot && slots.includes(this.lastAutoSlot)) {
      return this.load(this.lastAutoSlot);
    }

    let latestSlot = first;
    let latestIso = '';
    for (const slot of slots) {
      const loaded = await this.load(slot);
      if (loaded && loaded.meta.savedAtIso > latestIso) {
        latestIso = loaded.meta.savedAtIso;
        latestSlot = slot;
      }
    }
    return this.load(latestSlot);
  }

  private key(slot: string): string {
    return `save:${slot}`;
  }

  private async ensureLoaded(): Promise<void> {
    if (this.loaded) {
      return;
    }
    this.loaded = true;
    this.totalBytes = 0;
    this.bytes.clear();
    const keys = await this.kv.keys();
    for (const key of keys) {
      const text = await this.kv.get(key);
      if (text !== null) {
        const size = byteLength(text);
        this.bytes.set(key, size);
        this.totalBytes += size;
      }
    }
  }
}

import { App } from '@capacitor/app';
import { Capacitor } from '@capacitor/core';

export type LifecycleEventName = 'pause' | 'resume' | 'back';

export interface LifecycleSource {
  addEventListener(type: string, listener: () => void): void;
  removeEventListener?(type: string, listener: () => void): void;
  isHidden(): boolean;
}

export class Lifecycle {
  private readonly listeners = new Map<LifecycleEventName, Set<() => void>>();
  private paused = false;
  private readonly source: LifecycleSource;

  constructor(source: LifecycleSource) {
    this.source = source;
    source.addEventListener('visibilitychange', this.onVisibility);
    source.addEventListener('pause', this.onPause);
    source.addEventListener('resume', this.onResume);
    source.addEventListener('backButton', this.onBackButton);
  }

  on(event: LifecycleEventName, listener: () => void): () => void {
    const set = this.listeners.get(event) ?? new Set<() => void>();
    set.add(listener);
    this.listeners.set(event, set);
    return () => {
      set.delete(listener);
    };
  }

  destroy(): void {
    this.source.removeEventListener?.('visibilitychange', this.onVisibility);
    this.source.removeEventListener?.('pause', this.onPause);
    this.source.removeEventListener?.('resume', this.onResume);
    this.source.removeEventListener?.('backButton', this.onBackButton);
  }

  private emit(event: LifecycleEventName): void {
    for (const listener of this.listeners.get(event) ?? []) {
      listener();
    }
  }

  private readonly onVisibility = (): void => {
    if (this.source.isHidden()) {
      this.handlePause();
    } else {
      this.handleResume();
    }
  };

  private readonly onPause = (): void => {
    this.handlePause();
  };

  private readonly onResume = (): void => {
    this.handleResume();
  };

  private readonly onBackButton = (): void => {
    this.handlePause();
    this.emit('back');
  };

  private handlePause(): void {
    if (this.paused) {
      return;
    }
    this.paused = true;
    this.emit('pause');
  }

  private handleResume(): void {
    if (!this.paused) {
      return;
    }
    this.paused = false;
    this.emit('resume');
  }
}

export function bindLifecycleToBrowserAndCapacitor(): Lifecycle {
  const source: LifecycleSource = {
    addEventListener: (type, listener) => {
      if (type === 'visibilitychange') {
        document.addEventListener(type, listener);
        return;
      }
      if (type === 'pause' || type === 'resume' || type === 'backButton') {
        if (!Capacitor.isNativePlatform()) {
          return;
        }
        if (type === 'pause') {
          void App.addListener('pause', () => listener());
        } else if (type === 'resume') {
          void App.addListener('resume', () => listener());
        } else {
          void App.addListener('backButton', () => listener());
        }
      }
    },
    removeEventListener: (type, listener) => {
      if (type === 'visibilitychange') {
        document.removeEventListener(type, listener);
      }
    },
    isHidden: () => document.hidden,
  };

  return new Lifecycle(source);
}

export interface WatchdogReport {

export interface WatchdogOptions {

  onRecover: (report: WatchdogReport) => void;

interface ChannelState {
  lastBeat: number;
  freezeFired: boolean;
  recoverFired: boolean;
}

export class Watchdog {
  private readonly opts: WatchdogOptions;
  private readonly freezeMs: number;
  private readonly recoverMs: number;
  private readonly states: Record<WatchdogChannel, ChannelState>;
  private suspended = false;

  constructor(opts: WatchdogOptions) {
    this.opts = opts;
    this.freezeMs = opts.freezeMs ?? 2000;
    this.recoverMs = opts.recoverMs ?? 5000;
    const now = opts.now();
    this.states = {
      sim: { lastBeat: now, freezeFired: false, recoverFired: false },
      render: { lastBeat: now, freezeFired: false, recoverFired: false },
    };
  }

  beat(channel: WatchdogChannel): void {
    const state = this.states[channel];
    state.lastBeat = this.opts.now();
    state.freezeFired = false;
    state.recoverFired = false;
  }

  setSuspended(suspended: boolean): void {
    this.suspended = suspended;
    if (!suspended) {
      const now = this.opts.now();
      for (const channel of ['sim', 'render'] as const) {
        const state = this.states[channel];
        state.lastBeat = now;
        state.freezeFired = false;
        state.recoverFired = false;
      }
    }
  }

  check(): void {
    if (this.suspended) {
      return;
    }

    const now = this.opts.now();
    const stalled = (['sim', 'render'] as const).map((channel) => {
      const state = this.states[channel];
      return {
        channel,
        state,
        stalledMs: now - state.lastBeat,
      };
    });

    const freeze = stalled.find(
      (item) => item.stalledMs >= this.freezeMs && !item.state.freezeFired,
    );
    if (freeze) {
      for (const item of stalled) {
        if (item.stalledMs >= this.freezeMs) {
          item.state.freezeFired = true;
        }
      }
      this.opts.onFreeze({
        channel: freeze.channel,
        stalledMs: freeze.stalledMs,
        lastBeats: {
          sim: this.states.sim.lastBeat,
          render: this.states.render.lastBeat,
        },
      });
    }

    const recover = stalled.find(
      (item) => item.stalledMs >= this.recoverMs && !item.state.recoverFired,
    );
    if (recover) {
      for (const item of stalled) {
        if (item.stalledMs >= this.recoverMs) {
          item.state.recoverFired = true;
        }
      }
      this.opts.onRecover({
        channel: recover.channel,
        stalledMs: recover.stalledMs,
        lastBeats: {
          sim: this.states.sim.lastBeat,
          render: this.states.render.lastBeat,
        },
      });
    }
  }
}

export interface CapturedError {
  message: string;
  stack?: string;
  kind: 'error' | 'unhandledrejection';
}

export interface ErrorTarget {
  addEventListener(type: string, listener: (event: Event) => void): void;
  removeEventListener?(type: string, listener: (event: Event) => void): void;
}

export function installGlobalErrorCapture(
  target: ErrorTarget,
  onError: (error: CapturedError) => void,
): () => void {
  const errorHandler = (event: Event): void => {
    const errorEvent = event as ErrorEvent;
    const error = errorEvent.error;
    onError({
      message: errorEvent.message || (error instanceof Error ? error.message : 'Unknown error'),
      stack: error instanceof Error ? error.stack : undefined,
      kind: 'error',
    });
  };

  const rejectionHandler = (event: Event): void => {
    const rejection = event as PromiseRejectionEvent;
    const reason = rejection.reason;
    const error = reason instanceof Error ? reason : undefined;
    onError({
      message: error?.message ?? String(reason),
      stack: error?.stack,
      kind: 'unhandledrejection',
    });
  };

  target.addEventListener('error', errorHandler);
  target.addEventListener('unhandledrejection', rejectionHandler);

  return () => {
    target.removeEventListener?.('error', errorHandler);
    target.removeEventListener?.('unhandledrejection', rejectionHandler);
  };
}

    powerPreference: 'high-performance',
  });

  renderer.shadowMap.enabled = true;
  renderer.shadowMap.type = THREE.PCFSoftShadowMap;

export class RenderView {
  private readonly renderer: THREE.WebGLRenderer;
  private readonly scene = new THREE.Scene();
  private readonly camera: THREE.PerspectiveCamera;
  private readonly observer: ResizeObserver;
  private readonly loader: AssetLoader<THREE.Object3D, THREE.Texture, ArrayBuffer>;
  private readonly loadedModels: string[] = [];
  private frame = 0;
  private disposed = false;

  constructor(canvas: HTMLCanvasElement, registry: AssetRegistry) {
    this.renderer = createRenderer(canvas);
    this.camera = new THREE.PerspectiveCamera(55, 1, 0.1, 1000);
    this.camera.position.set(35, 40, 35);
    this.camera.lookAt(0, 0, 0);

    this.scene.background = new THREE.Color('#090D16');
    this.buildScene();

    this.observer = new ResizeObserver(() => {
      this.resize(canvas);
    });
    this.observer.observe(canvas);
    this.resize(canvas);

    const baseUrl = new URL('.', document.baseURI).toString();
    this.loader = new AssetLoader<THREE.Object3D, THREE.Texture, ArrayBuffer>({
      registry,
      baseUrl,
      decoders: createThreeDecoders(),
      fetchBytes: async (url) => {
        const response = await fetch(url);
        if (!response.ok) {
          throw new Error(`HTTP ${response.status} for ${url}`);
        }
        return response.arrayBuffer();
      },
    });

    void this.loadTank();
  }

  render(alpha: number): void {
    void alpha;
    this.frame += 1;
    this.renderer.render(this.scene, this.camera);
  }

  stats(): RenderStats {
    let meshCount = 0;
    this.scene.traverse((object) => {
      if (object instanceof THREE.Mesh) {
        meshCount += 1;
      }
    });
    return {
      loadedModels: [...this.loadedModels],
      meshCount,
      drawCalls: this.renderer.info.render.calls,
      frame: this.frame,
    };
  }

  dispose(): void {
    if (this.disposed) {
      return;
    }
    this.disposed = true;
    this.observer.disconnect();
    this.renderer.dispose();
  }

  private buildScene(): void {
    const hemisphere = new THREE.HemisphereLight(0xffffff, 0x223322, 1.2);
    this.scene.add(hemisphere);

    const directional = new THREE.DirectionalLight(0xffffff, 2.2);
    directional.position.set(30, 50, 20);
    directional.castShadow = true;
    directional.shadow.mapSize.set(2048, 2048);
    directional.shadow.camera.left = -100;
    directional.shadow.camera.right = 100;
    directional.shadow.camera.top = 100;
    directional.shadow.camera.bottom = -100;
    this.scene.add(directional);

    const ground = new THREE.Mesh(
      new THREE.PlaneGeometry(200, 200),
      new THREE.MeshStandardMaterial({ color: 0x24302a }),
    );
    ground.rotation.x = -Math.PI / 2;
    ground.receiveShadow = true;
    this.scene.add(ground);
  }

  private resize(canvas: HTMLCanvasElement): void {
    const width = Math.max(1, canvas.clientWidth);
    const height = Math.max(1, canvas.clientHeight);
    this.renderer.setSize(width, height, false);
    this.camera.aspect = width / height;
    this.camera.updateProjectionMatrix();
  }

  private async loadTank(): Promise<void> {
    try {
      const asset = await this.loader.load(TANK_ASSET_ID);
      if (this.disposed) {
        return;
      }
      if (asset instanceof THREE.Object3D) {
        this.scene.add(asset);
        this.loadedModels.push(TANK_ASSET_ID);
      } else {
        this.scene.add(fallbackModel());
      }
    } catch {
      if (!this.disposed) {
        this.scene.add(fallbackModel());
      }
    }
  }
}

export class ScreenRouter {
  private readonly root: HTMLElement;
  private readonly screens: Record<ScreenId, ScreenModule>;
  private readonly contextFactory: (router: ScreenRouter) => ScreenContext;
  private context: ScreenContext | null = null;
  private base: MountedScreen | null = null;
  private overlays: MountedScreen[] = [];

  constructor(
    root: HTMLElement,
    screens: Record<ScreenId, ScreenModule>,
    contextFactory: (router: ScreenRouter) => ScreenContext,
  ) {
    this.root = root;
    this.screens = screens;
    this.contextFactory = contextFactory;
  }

  start(id: ScreenId = 'title'): void {
    this.mountBase(id);
  }

  current(): ScreenId {
    return this.overlays.at(-1)?.id ?? this.base?.id ?? 'title';
  }

  go(id: ScreenId): void {
    const current = this.current();
    if (id === current) {
      return;
    }

    const overlay = this.isOverlayTarget(id, current);
    if (!this.canGo(current, id)) {
      this.reject(id);
      return;
    }

    if (overlay) {
      this.mountOverlay(id);
    } else {
      this.mountBase(id);
    }
  }

  back(): void {
    if (this.overlays.length > 0) {
      this.popOverlay();
      return;
    }
    if (!this.base) {
      return;
    }
    if (this.base.id === 'inGame') {
      this.go('pause');
      return;
    }
    const parent = PARENT[this.base.id];
    if (parent) {
      this.go(parent);
    }
  }

  private isOverlayTarget(id: ScreenId, current: ScreenId): boolean {
    if (id === 'settings') {
      return current === 'pause' && this.base?.id === 'inGame';
    }
    return OVERLAY_IDS.has(id) && this.base?.id === 'inGame';
  }

  private canGo(from: ScreenId, to: ScreenId): boolean {
    return ALLOWED[from].includes(to);
  }

  private reject(id: ScreenId): void {
    const message = `Screen transition not allowed: ${id}`;
    if (import.meta.env.DEV) {
      throw new Error(message);
    }
    console.warn(message);
  }

  private mountBase(id: ScreenId): void {
    for (const overlay of [...this.overlays].reverse()) {
      this.unmountScreen(overlay);
    }
    this.overlays = [];
    if (this.base) {
      this.unmountScreen(this.base);
    }
    const mounted = this.createScreen(id, false);
    mounted.el.style.visibility = 'visible';
    this.base = mounted;
  }

  private mountOverlay(id: ScreenId): void {
    const top = this.currentMounted();
    if (top) {
      top.el.style.visibility = 'hidden';
    }
    const mounted = this.createScreen(id, true);
    mounted.el.style.visibility = 'visible';
    this.overlays.push(mounted);
  }

  private popOverlay(): void {
    const mounted = this.overlays.pop();
    if (!mounted) {
      return;
    }
    this.unmountScreen(mounted);
    const top = this.currentMounted();
    if (top) {
      top.el.style.visibility = 'visible';
    }
  }

  private currentMounted(): MountedScreen | null {
    return this.overlays.at(-1) ?? this.base;
  }

  private createScreen(id: ScreenId, overlay: boolean): MountedScreen {
    const el = document.createElement('div');
    el.className = `screen screen-${id}${overlay ? ' overlay' : ''}`;
    el.dataset.screenId = id;
    this.root.append(el);
    const screen = this.screens[id];
    const unmount = screen.mount(el, this.ensureContext());
    return { id, el, unmount };
  }

  private unmountScreen(mounted: MountedScreen): void {
    mounted.unmount();
    mounted.el.remove();
  }

  private ensureContext(): ScreenContext {
    if (!this.context) {
      this.context = this.contextFactory(this);
    }
    return this.context;
  }
}

    return { ...snapshot(this.world) };
  }

  sendCommand(command: unknown): boolean {
    const validated = validateCommand(command);
    if (!validated) {
      return false;
    }
    this.pendingCommands.push(validated);
    return true;
  }

  renderStats(): RenderStats {
    if (!this.renderView) {
      return { loadedModels: [], meshCount: 0, drawCalls: 0, frame: 0 };
    }
    return this.renderView.stats();
  }

  freezeSim(ms: number): void {
    this.freezeSimUntil = performance.now() + Math.max(0, ms);
  }

  private attachRenderer(canvas: HTMLCanvasElement): RendererHandle {
    try {
      const view = new RenderView(canvas, this.registry);
      this.renderView = view;
      this.startLoop();
      return {
        error: null,
        dispose: () => {
          if (this.renderView === view) {
            this.stopLoop();
            view.dispose();
            this.renderView = null;
          }
        },
      };
    } catch {
      return {
        error: '3D rendering is unavailable on this device.',
        dispose: () => undefined,
      };
    }
  }

  private readonly onKeyDown = (event: KeyboardEvent): void => {
    if (event.key === 'Escape') {
      this.router.back();
    }
  };

  private pause(): void {
    if (this.paused) {
      return;
    }
    this.paused = true;
    this.watchdog.setSuspended(true);
    this.stopLoop();
  }

  private resume(): void {
    if (!this.paused) {
      return;
    }
    this.paused = false;
    this.watchdog.setSuspended(false);
    this.startLoop();
  }

  private startLoop(): void {
    if (this.raf !== 0 || !this.renderView || this.paused) {
      return;
    }
    this.lastFrame = performance.now();
    this.checkTimer = window.setInterval(() => {
      this.watchdog.check();
    }, 250);
    this.raf = window.requestAnimationFrame(this.frame);
  }

  private stopLoop(): void {
    if (this.raf !== 0) {
      window.cancelAnimationFrame(this.raf);
      this.raf = 0;
    }
    if (this.checkTimer !== 0) {
      window.clearInterval(this.checkTimer);
      this.checkTimer = 0;
    }
  }

  private readonly frame = (now: number): void => {
    this.raf = 0;
    if (this.paused || !this.renderView) {
      return;
    }

    const elapsed = now - this.lastFrame;
    this.lastFrame = now;
    const frozen = now < this.freezeSimUntil;
    let alpha = 0;

    if (!frozen) {
      const result = this.fixed.advance(elapsed);
      alpha = result.alpha;
      if (result.steps > 0) {
        this.watchdog.beat('sim');
      }
    }

    this.watchdog.beat('render');
    this.watchdog.check();
    this.renderView.render(alpha);
    this.raf = window.requestAnimationFrame(this.frame);
  };

  private stepSimulation(): void {
    const nextTick = this.world.tick + 1;
    const due = this.pendingCommands.filter((command) => command.tick === nextTick);
    this.world = step(this.world, due);
    this.pendingCommands = this.pendingCommands.filter((command) => command.tick > nextTick);
  }
}

export function createApp(root: HTMLElement): App {
  return new App(root);
}

export async function assertButtonLayout(page: Page): Promise<void> {

export function countDistinctPngColors(buffer: Buffer): number {
  const signature = buffer.subarray(0, 8).toString('hex');
  if (signature !== '89504e470d0a1a0a') {
    throw new Error('Not a PNG');
  }

  let offset = 8;
  let width = 0;
  let height = 0;
  let bitDepth = 0;
  let colorType = 0;
  const idat: Buffer[] = [];

  while (offset < buffer.length) {
    const length = buffer.readUInt32BE(offset);
    const type = buffer.toString('ascii', offset + 4, offset + 8);
    const data = buffer.subarray(offset + 8, offset + 8 + length);
    if (type === 'IHDR') {
      width = data.readUInt32BE(0);
      height = data.readUInt32BE(4);
      bitDepth = data[8] ?? 0;
      colorType = data[9] ?? 0;
    } else if (type === 'IDAT') {
      idat.push(data);
    } else if (type === 'IEND') {
      break;
    }
    offset += 12 + length;
  }

  if (bitDepth !== 8 || (colorType !== 6 && colorType !== 2)) {
    throw new Error(`Unsupported PNG format: bitDepth=${bitDepth} colorType=${colorType}`);
  }

  const bpp = colorType === 6 ? 4 : 3;
  const stride = width * bpp;
  const raw = inflateSync(Buffer.concat(idat));
  const out = Buffer.alloc(stride * height);
  let pos = 0;

  for (let y = 0; y < height; y += 1) {
    const filter = raw[pos];
    pos += 1;
    if (filter === undefined) {
      throw new Error('Truncated PNG');
    }
    const rowStart = y * stride;
    for (let x = 0; x < stride; x += 1) {
      const rawByte = raw[pos];
      pos += 1;
      if (rawByte === undefined) {
        throw new Error('Truncated PNG');
      }
      const left = x >= bpp ? out[rowStart + x - bpp] ?? 0 : 0;
      const up = y > 0 ? out[rowStart - stride + x] ?? 0 : 0;
      const upLeft = y > 0 && x >= bpp ? out[rowStart - stride + x - bpp] ?? 0 : 0;
      let value = 0;
      if (filter === 0) {
        value = rawByte;
      } else if (filter === 1) {
        value = rawByte + left;
      } else if (filter === 2) {
        value = rawByte + up;
      } else if (filter === 3) {
        value = rawByte + Math.floor((left + up) / 2);
      } else if (filter === 4) {
        const p = left + up - upLeft;
        const pa = Math.abs(p - left);
        const pb = Math.abs(p - up);
        const pc = Math.abs(p - upLeft);
        const predictor = pa <= pb && pa <= pc ? left : pb <= pc ? up : upLeft;
        value = rawByte + predictor;
      } else {
        throw new Error(`Unsupported PNG filter ${filter}`);
      }
      out[rowStart + x] = value & 0xff;
    }
  }

  const colors = new Set<string>();
  for (let y = 0; y < height; y += 1) {
    for (let x = 0; x < width; x += 1) {
      const index = y * stride + x * bpp;
      colors.add(`${out[index] ?? 0},${out[index + 1] ?? 0},${out[index + 2] ?? 0}`);
    }
  }
  return colors.size;
}

import { execFileSync, spawn } from 'node:child_process';
import { mkdtempSync, readFileSync, readdirSync, rmSync, statSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { expect, test } from '@playwright/test';

function walk(dir: string): string[] {
  const files: string[] = [];
  for (const name of readdirSync(dir)) {
    const full = join(dir, name);
    const info = statSync(full);
    if (info.isDirectory()) {
      files.push(...walk(full));
    } else {
      files.push(full);
    }
  }
  return files;
}

async function waitForUrl(url: string): Promise<void> {
  for (let i = 0; i < 80; i += 1) {
    try {
      const response = await fetch(url);
      if (response.ok) {
        return;
      }
    } catch {
      // Server may still be starting.
    }
    await new Promise((resolve) => setTimeout(resolve, 250));
  }
  throw new Error(`Server did not start at ${url}`);
}

# TankCommand

TankCommand is a mobile and desktop real-time strategy game rebuild, currently containing only the project foundation: deterministic simulation plumbing, asset and data containers, platform storage/lifecycle services, a DOM UI shell, and a three.js render shell. Gameplay rules, balance data, AI, pathfinding, combat and missions are not implemented yet.

## Prerequisites

- Node.js 22
- npm
- For Android packaging: Android Studio and the Capacitor Android platform generated after install.

## Scripts

| Script | Purpose |
| --- | --- |
| `npm run dev` | Start the Vite development server. |
| `npm run build` | Typecheck both TypeScript configs and build `dist/`. |
| `npm run preview` | Preview the built app on port 4173. |
| `npm run typecheck` | Run both TypeScript project typechecks. |
| `npm run lint` | Run ESLint with zero warnings allowed. |
| `npm test` | Run Vitest unit tests. |
| `npm run test:e2e` | Run Playwright end-to-end tests. |
| `npm run credits` | Regenerate `CREDITS.md` and `public/CREDITS.md`. |
| `npm run credits:check` | Fail if credits files are stale. |
| `npm run coverage:req` | Check requirement IDs against test titles. |
| `npm run cap:sync` | Build and sync the Capacitor Android project. |
| `npm run ci` | Run the full local CI sequence. |

## Folder map

- `src/sim/` — deterministic headless simulation plumbing.
- `src/data/` — unit data schema and placeholder rows.
- `src/assets/` — asset manifest schema, registry, loader and three.js decoders.
- `src/platform/` — storage, saves, lifecycle and watchdog services.
- `src/render/` — three.js renderer shell.
- `src/ui/` — DOM screen router, controls and screens.
- `src/testhook/` — browser test hook, included only in test builds.
- `tests/unit/` — Vitest tests.
- `tests/e2e/` — Playwright tests.
- `scripts/` — credits and requirement-coverage scripts.
- `docs/` — requirements, decisions and progress notes.

## Running end-to-end tests

Install Playwright Chromium once:

```sh
npx playwright install --with-deps chromium
```

Then run:

```sh
npm run test:e2e
```

The Playwright web server builds with `VITE_TEST_HOOK=1` and runs the preview server on port 4173.

## Adding an asset

1. Put the file under `public/assets/...`.
2. Add an entry to `public/assets/manifest.json` with the exact byte size and credit fields.
3. Run `npm run credits`.
4. Run `npm test` and `npm run credits:check`.

## Building Android

```sh
npx cap add android
npm run cap:sync
```

Then open the generated `android/` folder in Android Studio. The `android/` folder is generated by Capacitor and is not written by this package.

## 2. Self-review pass

Defects found and fixed during construction/re-read:

- The legacy `index.html` needed both a delete marker and the new Vite entry. I emitted a delete marker before the replacement file so the new entry is the final state.
- The initial router design would have treated `settings` opened from pause as a replacement screen. I special-cased `settings` as an overlay only when the current screen is `pause` and the base is `inGame`.
- Overlays could have left underlying buttons visible to Playwright, causing false overlap failures. The router now sets lower screens to `visibility: hidden` while an overlay is on top.
- A naive watchdog that reports every stalled channel would fire twice when both sim and render are idle. The implementation suppresses simultaneous channel reports for the same threshold crossing while still allowing a later episode after a beat.
- The test hook must not be present in production. It is only a conditional dynamic import behind `__TEST_HOOK__`, and the production e2e test checks both built JS text and the served window.
- `src/sim` must not use DOM/Node globals. The sim tsconfig uses only `ES2022`, and the ESLint restricted block covers imports, globals, `Math.random`, `Date.now` and `performance`.
- The credits screen needs a browser-fetchable copy. The credits script writes both `CREDITS.md` and `public/CREDITS.md` when using default paths, and `credits:check` checks both.
- The `AssetLoader` cache-entry creation needed to avoid `any` while assigning a promise after constructing the entry. It uses a `Partial<CacheEntry>` plus a narrow cast.
- The save envelope decode path needed to reject schema 999 before checksum validation, and to wrap all parsing in a catch so it never throws. Both are implemented.
- The UI inventory was reduced to the controls that screens actually register, and the e2e flow explicitly clicks every listed control, including the event-only recovery buttons.

Known risks that remain:

- I cannot run the code. The largest remaining uncertainty is exact API surface for `@capacitor/filesystem` `readdir` result typing; I cast the entries to a string-or-`{name}` union so both common shapes typecheck. Native storage is not unit tested, as allowed.
- The PNG decoder in `tests/e2e/helpers.ts` assumes Playwright canvas screenshots are 8-bit RGB/RGBA non-interlaced PNGs. That is the normal Playwright output, but it is the most brittle e2e helper.
- `npm run lint` may surface a type-aware ESLint rule complaint I did not anticipate. The code intentionally avoids `any`, uses `void` for intentionally ignored promises, and keeps restricted rules only on `src/sim`.

## 4. Deviations

- The legacy `index.html` is both marked for deletion and emitted as the new Vite entry. The spec asked for a delete marker while also requiring the new Vite entry at the same path, so the replacement file is the final state.
- The `settings` screen is an overlay only when entered from `pause`; from `title` it replaces the title screen. This is required by the overlay/back acceptance flow.
- The `recovery` buttons emit custom events and do not themselves navigate, as the spec says they only emit events in this package.
- Extra harmless API surface was added for testing and clarity: `AssetLoader.isCached`, `ScreenRouter.current`, `App.snapshot`, and public `migrations`.
